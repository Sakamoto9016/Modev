#include "main.h"
#include "audio_raw.h"

#include "driver/i2s_std.h"

#define I2S_BCLK  ((gpio_num_t)I2sBitClock)
#define I2S_LRCLK ((gpio_num_t)I2sLRClock)
#define I2S_DOUT  ((gpio_num_t)I2sOutput)

#define SAMPLE_RATE       16000
#define AUDIO_BUFFER_SIZE 256

i2s_chan_handle_t tx_handle;


// ============================================================
// I2S initialization
// ============================================================

void setupI2S()
{
    i2s_chan_config_t chan_cfg =
        I2S_CHANNEL_DEFAULT_CONFIG(
            I2S_NUM_0,
            I2S_ROLE_MASTER
        );

    ESP_ERROR_CHECK(
        i2s_new_channel(
            &chan_cfg,
            &tx_handle,
            nullptr
        )
    );


    i2s_std_config_t cfg = {

        .clk_cfg =
            I2S_STD_CLK_DEFAULT_CONFIG(
                SAMPLE_RATE
            ),

        .slot_cfg =
            I2S_STD_PHILIPS_SLOT_DEFAULT_CONFIG(
                I2S_DATA_BIT_WIDTH_16BIT,
                I2S_SLOT_MODE_MONO
            ),

        .gpio_cfg = {
            .mclk = I2S_GPIO_UNUSED,
            .bclk = I2S_BCLK,
            .ws   = I2S_LRCLK,
            .dout = I2S_DOUT,
            .din  = I2S_GPIO_UNUSED
        }
    };


    ESP_ERROR_CHECK(
        i2s_channel_init_std_mode(
            tx_handle,
            &cfg
        )
    );


    ESP_ERROR_CHECK(
        i2s_channel_enable(
            tx_handle
        )
    );
}


// ============================================================
// Blocking PCM player
// ============================================================

void playPCM(
    const int16_t *pcm,
    size_t samples
)
{
    int16_t buffer[AUDIO_BUFFER_SIZE];


    while (samples > 0) {

        size_t count =
            min(
                samples,
                (size_t)AUDIO_BUFFER_SIZE
            );


        // Apply volume.

        for (size_t i = 0; i < count; i++) {

            int32_t value =
                (int32_t)(
                    pcm[i] * I2sAmplifier
                );


            if (value > 32767)
                value = 32767;

            if (value < -32768)
                value = -32768;


            buffer[i] =
                (int16_t)value;
        }


        size_t written = 0;


        esp_err_t err =
            i2s_channel_write(
                tx_handle,
                buffer,
                count * sizeof(int16_t),
                &written,
                portMAX_DELAY
            );


        if (err != ESP_OK) {

            return;
        }


        size_t consumed =
            written / sizeof(int16_t);


        pcm += consumed;
        samples -= consumed;
    }
}


// ============================================================
// Audio task
// ============================================================

void audioTask(void *parameter)
{
    const int16_t *pcm =
        reinterpret_cast<const int16_t *>(
            output_pcm
        );


    size_t samples =
        output_pcm_len /
        sizeof(int16_t);


    playPCM(
        pcm,
        samples
    );


    vTaskDelete(nullptr);
}


// ============================================================
// Setup
// ============================================================

void setup()
{
    sys.begin();
    gfx.begin();


    gfx.clearQuick(0);

    gfx.setFont("6x12");


    gfx.text(
        "I2S 16K PCM",
        0,
        0,
        0xFFFF
    );

    gfx.flush();


    setupI2S();


    size_t samples =
        output_pcm_len /
        sizeof(int16_t);


    char buf[64];

    sprintf(
        buf,
        "%u samples",
        (unsigned)samples
    );


    gfx.clear(0);


    gfx.text(
        "PLAYING",
        0,
        0,
        0xFFFF
    );


    gfx.text(
        buf,
        0,
        20,
        0xFFFF
    );


    gfx.text(
        "16kHz MONO",
        0,
        40,
        0xFFFF
    );


    gfx.flush();


    // Create independent audio task.

    xTaskCreate(
        audioTask,
        "AudioTask",
        4096,
        nullptr,
        1,
        nullptr
    );
}
